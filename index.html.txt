<script>

  /* =====================================================
     PARAMETER DARI DAHAGA POS
  ===================================================== */

  const params =
    new URLSearchParams(
      window.location.search
    );

  const targetOrigin =
    params.get("targetOrigin") || "*";

  const sessionId =
    params.get("session") || "";


  /* =====================================================
     SCANNER
  ===================================================== */

  let html5QrCode = null;

  let sedangScan = false;

  let barcodeSudahTerbaca = false;


  /* =====================================================
     MULAI SCAN
  ===================================================== */

  async function mulaiScan() {

    if (sedangScan) {
      return;
    }

    barcodeSudahTerbaca = false;

    const status =
      document.getElementById("status");

    const btnStart =
      document.getElementById("btnStart");

    const btnStop =
      document.getElementById("btnStop");


    try {

      status.textContent =
        "?? Meminta izin kamera...";


      if (!html5QrCode) {

        html5QrCode =
          new Html5Qrcode("reader");

      }


      const config = {

        fps: 10,

        qrbox: function(
          viewfinderWidth,
          viewfinderHeight
        ) {

          const ukuran =
            Math.min(
              viewfinderWidth,
              viewfinderHeight
            ) * 0.70;

          return {
            width: ukuran,
            height: ukuran
          };

        },

        aspectRatio: 1.0

      };


      await html5QrCode.start(

        {
          facingMode: "environment"
        },

        config,

        ketikaBarcodeTerbaca,

        ketikaScanGagal

      );


      sedangScan = true;

      btnStart.disabled = true;

      btnStop.disabled = false;

      status.textContent =
        "?? Kamera aktif. Arahkan ke barcode menu.";

    }

    catch (error) {

      console.error(
        "Gagal membuka kamera:",
        error
      );

      status.textContent =
        "? Kamera tidak dapat digunakan.";

      alert(
        "Kamera tidak dapat digunakan.\n\n" +

        "Pastikan:\n" +

        "1. Browser mengizinkan kamera.\n" +

        "2. Scanner dibuka melalui HTTPS.\n" +

        "3. Kamera tidak sedang digunakan aplikasi lain.\n\n" +

        "Detail:\n" +

        error
      );

      sedangScan = false;

      btnStart.disabled = false;

      btnStop.disabled = true;

    }

  }


  /* =====================================================
     BARCODE TERBACA
  ===================================================== */

  async function ketikaBarcodeTerbaca(
    decodedText,
    decodedResult
  ) {

    if (barcodeSudahTerbaca) {
      return;
    }

    barcodeSudahTerbaca = true;


    const barcode =
      String(
        decodedText || ""
      ).trim();


    if (!barcode) {

      barcodeSudahTerbaca = false;

      return;

    }


    console.log(
      "Barcode terbaca:",
      barcode
    );


    document.getElementById(
      "hasil"
    ).textContent = barcode;


    document.getElementById(
      "status"
    ).textContent =
      "? Barcode berhasil dibaca.";


    /* =================================================
       KIRIM KE HALAMAN KASIR
    ================================================= */

    try {

      if (
        window.opener &&
        !window.opener.closed
      ) {

        window.opener.postMessage(

          {
            source:
              "DAHAGA_BARCODE_SCANNER",

            type:
              "BARCODE_RESULT",

            barcode:
              barcode,

            session:
              sessionId,

            waktu:
              Date.now()

          },

          targetOrigin

        );

      } else {

        alert(
          "Halaman Kasir tidak ditemukan.\n\n" +
          "Silakan buka scanner dari tombol SCAN di Kasir."
        );

      }

    }

    catch (error) {

      console.error(
        "Gagal mengirim barcode:",
        error
      );

      alert(
        "Barcode berhasil dibaca, tetapi gagal dikirim ke Kasir."
      );

    }


    /* =================================================
       STOP CAMERA
    ================================================= */

    await berhentiScan();


    /* =================================================
       TUTUP TAB SCANNER
    ================================================= */

    setTimeout(
      function() {

        try {

          window.close();

        }

        catch (error) {

          console.log(
            "Window close:",
            error
          );

        }

      },

      700

    );

  }


  /* =====================================================
     ERROR SCAN
  ===================================================== */

  function ketikaScanGagal(
    errorMessage
  ) {

    /*
      Jangan tampilkan error setiap frame
      karena scanner memang akan terus mencoba
      membaca barcode.
    */

  }


  /* =====================================================
     STOP SCANNER
  ===================================================== */

  async function berhentiScan() {

    const btnStart =
      document.getElementById("btnStart");

    const btnStop =
      document.getElementById("btnStop");

    const status =
      document.getElementById("status");


    if (
      html5QrCode &&
      sedangScan
    ) {

      try {

        await html5QrCode.stop();

      }

      catch (error) {

        console.error(
          "Gagal menghentikan scanner:",
          error
        );

      }

    }


    sedangScan = false;

    btnStart.disabled = false;

    btnStop.disabled = true;


    if (!barcodeSudahTerbaca) {

      status.textContent =
        "Scanner dihentikan.";

    }

  }


</script>
